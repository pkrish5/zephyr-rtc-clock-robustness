# Item R18

Target case: `test_system_workqueue_delay`.

## Module 1

```c
#include <zephyr/kernel.h>
#include <zephyr/sys/atomic.h>
#include <zephyr/ztest.h>

#define DELAY_MS 100
#define COMPLETION_TIMEOUT_MS 1000
#define SETTLE_MS 20

static struct k_work_delayable delayed_work;
K_SEM_DEFINE(work_done, 0, 1);
static atomic_t handler_count;
static int64_t schedule_time_ms;
static int64_t handler_time_ms;

static void delayed_work_handler(struct k_work *work)
{
	ARG_UNUSED(work);
	handler_time_ms = k_uptime_get();
	atomic_inc(&handler_count);
	k_sem_give(&work_done);
}

ZTEST(delayable_work, test_system_workqueue_delay)
{
	const int64_t one_tick_ms = k_ticks_to_ms_ceil64(1);

	atomic_set(&handler_count, 0);
	k_sem_reset(&work_done);
	k_work_init_delayable(&delayed_work, delayed_work_handler);

	schedule_time_ms = k_uptime_get();
	zassert_equal(k_work_schedule(&delayed_work, K_MSEC(DELAY_MS)), 1,
		      "Scheduling the delayable work should succeed");
	zassert_true(k_work_delayable_is_pending(&delayed_work),
		     "Work should be pending immediately after scheduling");

	zassert_ok(k_sem_take(&work_done, K_MSEC(COMPLETION_TIMEOUT_MS)),
		  "Work handler did not complete before the timeout");

	zassert_true(handler_time_ms - schedule_time_ms >=
		     DELAY_MS - one_tick_ms,
		     "Handler ran before the requested delay, allowing one tick of granularity");
	zassert_equal(atomic_get(&handler_count), 1,
		      "Handler should execute exactly once");
	zassert_false(k_work_delayable_is_pending(&delayed_work),
		      "Completed work should no longer be pending");

	k_msleep(SETTLE_MS);
	zassert_equal(atomic_get(&handler_count), 1,
		      "Handler should not execute again");
	zassert_false(k_work_delayable_is_pending(&delayed_work),
		      "Work should remain not pending after completion");
}

ZTEST_SUITE(delayable_work, NULL, NULL, NULL, NULL, NULL);
```

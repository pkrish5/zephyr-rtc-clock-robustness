# Item R28

Target case: `test_calendar_counts_across_year_boundary`.

## Module 1

```c
#include <stdint.h>

#include <zephyr/device.h>
#include <zephyr/devicetree.h>
#include <zephyr/drivers/rtc.h>
#include <zephyr/kernel.h>
#include <zephyr/ztest.h>

#define RTC_DEVICE DEVICE_DT_GET(DT_ALIAS(rtc))

static int64_t days_from_civil(int year, unsigned int month, unsigned int day)
{
	year -= month <= 2U;
	int era = (year >= 0 ? year : year - 399) / 400;
	unsigned int year_of_era = (unsigned int)(year - era * 400);
	unsigned int day_of_year =
		(153U * (month + (month > 2U ? (unsigned int)-3 : 9U)) + 2U) / 5U +
		day - 1U;
	unsigned int day_of_era = year_of_era * 365U + year_of_era / 4U -
				  year_of_era / 100U + day_of_year;

	return (int64_t)era * 146097 + (int64_t)day_of_era - 719468;
}

static int64_t rtc_seconds(const struct rtc_time *time)
{
	int year = time->tm_year + 1900;
	unsigned int month = (unsigned int)time->tm_mon + 1U;

	return days_from_civil(year, month, (unsigned int)time->tm_mday) * 86400 +
	       (int64_t)time->tm_hour * 3600 +
	       (int64_t)time->tm_min * 60 + time->tm_sec;
}

ZTEST(rtc_calendar, test_calendar_counts_across_year_boundary)
{
	const struct device *rtc = RTC_DEVICE;
	struct rtc_time initial = {
		.tm_sec = 50,
		.tm_min = 59,
		.tm_hour = 23,
		.tm_mday = 31,
		.tm_mon = 11,
		.tm_year = 124,
	};
	struct rtc_time observed;

	zassert_true(device_is_ready(rtc), "RTC device is not ready");
	zassert_equal(rtc_set_time(rtc, &initial), 0, "Failed to set RTC calendar");
	zassert_equal(rtc_get_time(rtc, &observed), 0, "Failed to read initial RTC calendar");
	zassert_equal(rtc_seconds(&observed), rtc_seconds(&initial),
		      "Initial RTC calendar did not match the value set");

	int64_t start = rtc_seconds(&observed);
	int64_t previous = start;
	int64_t target = start + 10;
	int64_t deadline = k_uptime_get() + 30000;

	while (k_uptime_get() < deadline) {
		k_sleep(K_MSEC(100));
		zassert_equal(rtc_get_time(rtc, &observed), 0, "Failed to read RTC calendar");

		int64_t current = rtc_seconds(&observed);

		zassert_true(current >= previous, "RTC calendar moved backward");
		previous = current;
		if (current >= target) {
			break;
		}
	}

	zassert_true(previous >= target,
		     "RTC did not advance ten calendar seconds within 30 seconds");
	zassert_true(observed.tm_year > initial.tm_year,
		     "RTC calendar did not cross the year boundary");
}

ZTEST_SUITE(rtc_calendar, NULL, NULL, NULL, NULL, NULL);
```

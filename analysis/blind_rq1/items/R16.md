# Item R16

Target case: `test_one_shot_timer_wakes_blocked_thread`.

## Module 1

```c
#include <zephyr/kernel.h>
#include <zephyr/ztest.h>

#define TIMER_DELAY K_MSEC(100)

K_TIMER_DEFINE(test_timer, NULL, NULL);
K_SEM_DEFINE(completion, 0, 1);
K_THREAD_STACK_DEFINE(worker_stack, 1024);
static struct k_thread worker_thread;
static volatile uint32_t received_status;

static void timer_worker(void *a, void *b, void *c)
{
	ARG_UNUSED(a);
	ARG_UNUSED(b);
	ARG_UNUSED(c);

	received_status = k_timer_status_sync(&test_timer);
	k_sem_give(&completion);
}

ZTEST(timer_interrupt, test_one_shot_timer_wakes_blocked_thread)
{
	received_status = 0;
	k_sem_reset(&completion);

	k_timer_stop(&test_timer);
	k_timer_start(&test_timer, TIMER_DELAY, K_NO_WAIT);
	k_thread_create(&worker_thread, worker_stack, K_THREAD_STACK_SIZEOF(worker_stack),
			 timer_worker, NULL, NULL, NULL, K_PRIO_PREEMPT(1), 0, K_NO_WAIT);

	k_sleep(K_MSEC(10));
	zassert_equal(k_thread_state_str(&worker_thread)[0], 'p',
		      "worker should be pending on the timer status");
	zassert_equal(k_sem_take(&completion, K_NO_WAIT), -EAGAIN,
		      "completion must not be signaled before expiry");

	zassert_equal(k_sem_take(&completion, K_SECONDS(2)), 0,
		      "timer expiry should signal completion");
	zassert_equal(received_status, 1U,
		      "worker should receive exactly one timer expiration");
	zassert_equal(k_timer_status_get(&test_timer), 0U,
		      "timer expiration status should have been consumed");

	k_thread_join(&worker_thread, K_SECONDS(1));
	k_timer_stop(&test_timer);
}

ZTEST_SUITE(timer_interrupt, NULL, NULL, NULL, NULL, NULL);
```

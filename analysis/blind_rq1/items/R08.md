# Item R08

Target case: `test_isr_wakes_sleeping_thread_early`.

## Module 1

```c
#include <zephyr/kernel.h>
#include <zephyr/irq_offload.h>
#include <zephyr/ztest.h>

#define SLEEP_DURATION K_SECONDS(5)
#define WAKE_DELAY_MS 100
#define WAKE_MARGIN_MS 500

K_THREAD_STACK_DEFINE(sleeper_stack, 1024);
static struct k_thread sleeper_thread;
static struct k_sem resumed_sem;
static k_tid_t sleeper_tid;
static int64_t sleep_started_ms;
static int64_t resumed_ms;
static int64_t wakeup_requested_ms;

static void wakeup_isr(const void *unused)
{
	ARG_UNUSED(unused);

	wakeup_requested_ms = k_uptime_get();
	k_wakeup(sleeper_tid);
}

static void sleeper_entry(void *p1, void *p2, void *p3)
{
	ARG_UNUSED(p1);
	ARG_UNUSED(p2);
	ARG_UNUSED(p3);

	sleep_started_ms = k_uptime_get();
	(void)k_sleep(SLEEP_DURATION);
	resumed_ms = k_uptime_get();
	k_sem_give(&resumed_sem);
}

ZTEST(wakeup_test, test_isr_wakes_sleeping_thread_early)
{
	int ret;

	k_sem_init(&resumed_sem, 0, 1);
	sleep_started_ms = 0;
	resumed_ms = 0;
	wakeup_requested_ms = 0;

	sleeper_tid = k_thread_create(&sleeper_thread, sleeper_stack,
				      K_THREAD_STACK_SIZEOF(sleeper_stack),
				      sleeper_entry, NULL, NULL, NULL,
				      K_PRIO_PREEMPT(1), 0, K_NO_WAIT);
	zassert_not_null(sleeper_tid, "Failed to create sleeper thread");

	while (sleep_started_ms == 0) {
		k_yield();
	}
	k_msleep(WAKE_DELAY_MS);
	irq_offload(wakeup_isr, NULL);

	ret = k_sem_take(&resumed_sem, K_SECONDS(2));
	zassert_equal(ret, 0, "Sleeper did not signal completion after wakeup");
	zassert_true(wakeup_requested_ms >= sleep_started_ms,
		     "ISR wakeup request was not recorded after sleep began");
	zassert_true(resumed_ms >= wakeup_requested_ms,
		     "Sleeper completion preceded ISR wakeup request");
	zassert_true(resumed_ms - sleep_started_ms <
		     (int64_t)k_ticks_to_ms_floor32(SLEEP_DURATION.ticks) - WAKE_MARGIN_MS,
		     "Observed sleep duration was not shorter than requested with margin");
}

ZTEST_SUITE(wakeup_test, NULL, NULL, NULL, NULL, NULL);
```

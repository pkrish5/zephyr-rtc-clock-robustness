# Item R02

Target case: `test_both_alarm_channels`.

## Module 1

```c
#include <zephyr/device.h>
#include <zephyr/drivers/rtc.h>
#include <zephyr/kernel.h>
#include <zephyr/sys/atomic.h>
#include <zephyr/sys/util.h>
#include <zephyr/ztest.h>

#define ALARM_A 0U
#define ALARM_B 1U
#define ALARM_BITS (BIT(ALARM_A) | BIT(ALARM_B))
#define ALARM_FIELDS (RTC_ALARM_TIME_MASK_SECOND | RTC_ALARM_TIME_MASK_MINUTE | \
		  RTC_ALARM_TIME_MASK_HOUR)

static atomic_t callback_ids;
static atomic_t callback_count;

static void alarm_callback(const struct device *dev, uint16_t id, void *user_data)
{
	ARG_UNUSED(dev);
	ARG_UNUSED(user_data);

	atomic_inc(&callback_count);
	if (id < 2U) {
		atomic_or(&callback_ids, BIT(id));
	}
}

ZTEST(rtc_alarm, test_both_alarm_channels)
{
	const struct device *rtc = DEVICE_DT_GET(DT_ALIAS(rtc));
	struct rtc_time start = {
		.tm_sec = 55,
		.tm_min = 59,
		.tm_hour = 23,
		.tm_mday = 31,
		.tm_mon = 11,
		.tm_year = 124,
	};
	struct rtc_time alarm = {
		.tm_sec = 5,
		.tm_min = 0,
		.tm_hour = 0,
		.tm_mday = 1,
		.tm_mon = 0,
		.tm_year = 125,
	};
	struct rtc_time now;
	int64_t elapsed_ms = 0;
	int ret;

	zassert_true(device_is_ready(rtc), "RTC device is not ready");
	atomic_set(&callback_ids, 0);
	atomic_set(&callback_count, 0);

	ret = rtc_set_time(rtc, &start);
	zassert_equal(ret, 0, "rtc_set_time failed: %d", ret);

	for (uint16_t id = ALARM_A; id <= ALARM_B; ++id) {
		ret = rtc_alarm_set_callback(rtc, id, alarm_callback, NULL);
		zassert_equal(ret, 0, "setting callback for alarm %u failed: %d", id, ret);
		ret = rtc_alarm_set_time(rtc, id, ALARM_FIELDS, &alarm);
		zassert_equal(ret, 0, "setting alarm %u failed: %d", id, ret);
	}

	while (elapsed_ms <= 15000) {
		ret = rtc_get_time(rtc, &now);
		zassert_equal(ret, 0, "rtc_get_time failed: %d", ret);

		if (now.tm_year > alarm.tm_year ||
		    (now.tm_year == alarm.tm_year && now.tm_mon > alarm.tm_mon) ||
		    (now.tm_year == alarm.tm_year && now.tm_mon == alarm.tm_mon &&
		     now.tm_mday > alarm.tm_mday) ||
		    (now.tm_year == alarm.tm_year && now.tm_mon == alarm.tm_mon &&
		     now.tm_mday == alarm.tm_mday && now.tm_hour > alarm.tm_hour) ||
		    (now.tm_year == alarm.tm_year && now.tm_mon == alarm.tm_mon &&
		     now.tm_mday == alarm.tm_mday && now.tm_hour == alarm.tm_hour &&
		     now.tm_min > alarm.tm_min) ||
		    (now.tm_year == alarm.tm_year && now.tm_mon == alarm.tm_mon &&
		     now.tm_mday == alarm.tm_mday && now.tm_hour == alarm.tm_hour &&
		     now.tm_min == alarm.tm_min && now.tm_sec >= alarm.tm_sec)) {
			break;
		}

		zassert_equal(atomic_get(&callback_count), 0,
			      "an RTC alarm callback fired before the alarm time");
		k_sleep(K_MSEC(20));
		elapsed_ms += 20;
	}
	zassert_true(elapsed_ms <= 15000, "RTC did not reach the alarm time within 15 seconds");

	for (int elapsed = 0; elapsed < 3000 &&
	     atomic_get(&callback_count) < 2; elapsed += 10) {
		k_sleep(K_MSEC(10));
	}

	zassert_equal(atomic_get(&callback_count), 2,
		      "expected exactly two alarm callbacks, got %d",
		      atomic_get(&callback_count));
	zassert_equal(atomic_get(&callback_ids), ALARM_BITS,
		      "callbacks did not report both configured alarm IDs");
}

ZTEST_SUITE(rtc_alarm, NULL, NULL, NULL, NULL, NULL);
```

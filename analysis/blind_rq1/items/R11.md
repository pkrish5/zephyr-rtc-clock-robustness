# Item R11

Target case: `test_isr_to_thread_fifo`.

## Module 1

```c
#include <zephyr/ztest.h>
#include <zephyr/kernel.h>
#include <zephyr/irq_offload.h>

struct fifo_item {
    void *reserved;
    int value;
};

K_FIFO_DEFINE(test_fifo);

static struct fifo_item item_a;
static struct fifo_item item_b;
static volatile bool isr_ran;

static void fifo_isr(void *unused)
{
    ARG_UNUSED(unused);

    isr_ran = true;
    k_fifo_put(&test_fifo, &item_a);
    k_fifo_put(&test_fifo, &item_b);
}

ZTEST(fifo_irq, test_isr_to_thread_fifo)
{
    item_a.value = 1;
    item_b.value = 2;
    isr_ran = false;

    irq_offload(fifo_isr, NULL);

    zassert_true(isr_ran, "FIFO handler did not run in interrupt context");
    zassert_equal(k_fifo_get(&test_fifo, K_NO_WAIT), &item_a,
                  "first FIFO item must be item_a");
    zassert_equal(k_fifo_get(&test_fifo, K_NO_WAIT), &item_b,
                  "second FIFO item must be item_b");
    zassert_is_null(k_fifo_get(&test_fifo, K_NO_WAIT),
                    "FIFO must be empty after both gets");
}

ZTEST_SUITE(fifo_irq, NULL, NULL, NULL, NULL, NULL);
```

# Item R20

Target case: `test_isr_to_thread_lifo`.

## Module 1

```c
#include <zephyr/ztest.h>
#include <zephyr/kernel.h>
#include <zephyr/irq_offload.h>

struct lifo_item {
    void *reserved;
    int value;
};

K_LIFO_DEFINE(test_lifo);

static struct lifo_item item_a;
static struct lifo_item item_b;
static volatile bool isr_ran;

static void lifo_put_isr(const void *unused)
{
    ARG_UNUSED(unused);
    isr_ran = true;
    k_lifo_put(&test_lifo, &item_a);
    k_lifo_put(&test_lifo, &item_b);
}

ZTEST(lifo_irq_transfer, test_isr_to_thread_lifo)
{
    void *first;
    void *second;

    isr_ran = false;
    irq_offload(lifo_put_isr, NULL);

    zassert_true(isr_ran, "irq_offload handler did not run");

    first = k_lifo_get(&test_lifo, K_NO_WAIT);
    second = k_lifo_get(&test_lifo, K_NO_WAIT);

    zassert_equal_ptr(first, &item_b, "first get should return last inserted item");
    zassert_equal_ptr(second, &item_a, "second get should return first inserted item");
    zassert_is_null(k_lifo_get(&test_lifo, K_NO_WAIT), "LIFO should be empty");
}

ZTEST_SUITE(lifo_irq_transfer, NULL, NULL, NULL, NULL, NULL);
```
